cmake_minimum_required(VERSION 3.26)
project(ctest_protocol NONE)
enable_testing()
add_test(NAME passed COMMAND /bin/sh -c "exit 0")
add_test(NAME failed COMMAND /bin/sh -c "echo original-marker; exit 7")
add_test(NAME skipped COMMAND /bin/sh -c "exit 77")
set_tests_properties(skipped PROPERTIES SKIP_RETURN_CODE 77)
add_test(NAME disabled COMMAND /bin/sh -c "exit 0")
set_tests_properties(disabled PROPERTIES DISABLED TRUE)
add_test(NAME timed_out COMMAND /usr/bin/sleep 20)
set_tests_properties(timed_out PROPERTIES TIMEOUT 0.05)
add_test(NAME crashed COMMAND /bin/sh -c "kill -SEGV $$")
add_test(NAME will_fail COMMAND /bin/sh -c "exit 3")
set_tests_properties(will_fail PROPERTIES WILL_FAIL TRUE)
add_test(NAME "parse[value]+edge" COMMAND /bin/sh -c "exit 0")
