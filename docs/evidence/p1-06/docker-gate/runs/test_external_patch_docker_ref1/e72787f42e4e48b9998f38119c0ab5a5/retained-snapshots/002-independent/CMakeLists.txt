cmake_minimum_required(VERSION 3.20)
project(repropatch_cpp_bugs LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(stats src/stats.cpp)
target_include_directories(stats PUBLIC src)
target_compile_options(stats PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(stats_test tests/stats_test.cpp)
target_link_libraries(stats_test PRIVATE stats)
enable_testing()
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/src/stats.cpp" omit REGEX "OMIT_REQUIRED_TEST")
foreach(case negative_values positive_values empty_values mixed_values)
  if(omit AND case STREQUAL "negative_values")
    continue()
  endif()
  add_test(NAME ${case} COMMAND stats_test ${case})
endforeach()
